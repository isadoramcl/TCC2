RR=dict(zip(['baixa','media','alta','muito_alta'],[1.,1.9018,2.8368,5.3670]))
def read(p):return pd.read_csv(p,float_precision='round_trip')
def sha(p):return hashlib.sha256(p.read_bytes()).hexdigest()
def setup():
 if not CACHE:
  p=S.carregar_parametros(OUT/'snapshot/config/parametros.yaml',OUT/'snapshot/config/parametros_derivados.yaml');op=yaml.safe_load((OUT/'snapshot/config/mvp.yaml').read_text())['opcoes'];op['instrumentar_tarefas']=False
  CACHE.update(p=p,op=op,inst=read(OUT/'snapshot/data/processed/psplib/instancias_j60.csv').set_index('arquivo'),nominal=read(OUT/'snapshot/data/processed/psplib/tarefas_j60_com_di.csv'))
 return CACHE

def job(arg):
 variant,group,arq,seed,cen=arg;c=setup();p=copy.deepcopy(c['p'])
 if variant=='tamanho':p['risco']['razoes_de_risco']['valor']=RR.copy()
 elif variant!='complexidade':
  weights=p['dificuldade']['pesos_Di']['varredura_alternativa'][variant];p['dificuldade']['pesos_Di']['valor']=weights
  if variant not in c:c[variant]=E.recalcular_di(c['nominal'],weights)
 frame=c['nominal'] if variant in ['complexidade','tamanho'] else c[variant];g=frame[frame.arquivo==arq].copy();ln=c['inst'].loc[arq]
 r=SimulacaoMVP(g,[int(x) for x in str(ln.disponibilidades).split(';')],int(ln.makespan_sem_recursos),p,cen,seed,opcoes=OpcoesMVP(**c['op'])).executar()
 row={k:v for k,v in asdict(r).items() if isinstance(v,(int,float,bool))};row.update(r.contadores);den=row['p1_omissao']+row['p3_analitica']
 row.update(variante=variant,conjunto=group,arquivo=arq,semente=seed,cenario=cen,violacoes=len(r.violacoes),mensagens=';'.join(r.violacoes),atraso_relativo=r.makespan/r.makespan_cpm,fracao_porta1=row['p1_omissao']/den if den else np.nan)
 return row

def contrasts(d):
 rows=[]
 for (var,group),g in d.groupby(['variante','conjunto']):
  for m in MET:
   p=g.pivot(index=['arquivo','semente'],columns='cenario',values=m);x=(p.adaptativa-p.centralizada).groupby('arquivo').mean();n=len(x);mean=float(x.mean());h=float(t.ppf(.975,n-1)*x.std(ddof=1)/np.sqrt(n))
   rows.append(dict(variante=var,conjunto=group,metrica=m,media=mean,IC_inf=mean-h,IC_sup=mean+h,N_instancias=n,negativo_IC_exclui_zero=bool(mean+h<0)))
 return pd.DataFrame(rows)

def execute(jobs,name):
 start=time.monotonic()
 with (OUT/(name+'.csv')).open('x') as f,ProcessPoolExecutor(4) as pool:
  writer=None
  for i,row in enumerate(pool.map(job,jobs,chunksize=1),1):
   if writer is None:writer=csv.DictWriter(f,fieldnames=list(row));writer.writeheader()
   writer.writerow(row);f.flush()
   if row['violacoes'] or not row['concluiu']:raise RuntimeError(f'Falha de validade, evidência preservada: {row}')
   if i%128==0 or i==len(jobs):print(name,i,'/',len(jobs),'s',round(time.monotonic()-start,1),flush=True)
 return read(OUT/(name+'.csv'))
